Supabase:\n\n```javascript\n/* ==========================================\n   BERITA MODULE - Supabase Integration\n   Membaca berita dari Supabase database\n   ========================================== */\n\nconst BeritaModule = (() => {\n  let currentFilter = 'Semua';\n  let beritaCache = [];\n\n  // Initialize module\n  async function init() {\n    try {\n      console.log('🔄 Initializing Berita module...');\n      \n      // Check Supabase\n      if (!window.supabaseClient) {\n        throw new Error('Supabase client not initialized');\n      }\n\n      // Load berita dari database\n      await loadBerita();\n      \n      // Setup event listeners\n      setupListeners();\n      \n      console.log('✅ Berita module initialized');\n    } catch (error) {\n      console.error('❌ Error initializing Berita:', error);\n      showToast('Error loading berita', 'error');\n    }\n  }\n\n  // Load berita dari Supabase\n  async function loadBerita() {\n    try {\n      console.log('📥 Loading berita from Supabase...');\n      \n      const { data, error } = await window.supabaseClient\n        .from('berita')\n        .select('*')\n        .order('date', { ascending: false });\n\n      if (error) throw error;\n\n      beritaCache = data || [];\n      console.log(`✅ Loaded ${beritaCache.length} berita`);\n      \n      // Render ke halaman\n      render();\n      \n      return beritaCache;\n    } catch (error) {\n      console.error('❌ Error loading berita:', error);\n      showToast('Gagal memuat berita', 'error');\n      return [];\n    }\n  }\n\n  // Get berita dengan filter kategori\n  function getBerita(category = 'Semua') {\n    if (category === 'Semua') return beritaCache;\n    return beritaCache.filter(b => b.category === category);\n  }\n\n  // Render berita ke halaman\n  function render() {\n    try {\n      const container = document.getElementById('berita-container');\n      if (!container) return;\n\n      const berita = getBerita(currentFilter);\n\n      if (berita.length === 0) {\n        container.innerHTML = '<p class=\"text-center text-muted\">Belum ada berita</p>';\n        return;\n      }\n\n      container.innerHTML = berita\n        .map(item => `\n          <div class=\"berita-card\">\n            ${item.image ? `<img src=\"${item.image}\" alt=\"${item.title}\">` : ''}\n            <h3>${item.title}</h3>\n            <p class=\"category\">${item.category}</p>\n            <p class=\"date\">${new Date(item.date).toLocaleDateString('id-ID')}</p>\n            <p>${item.content.substring(0, 100)}...</p>\n            <a href=\"berita-detail.html?id=${item.id}\" class=\"btn-read-more\">Baca Selengkapnya</a>\n          </div>\n        `)\n        .join('');\n    } catch (error) {\n      console.error('❌ Error rendering berita:', error);\n    }\n  }\n\n  // Filter berita by category\n  function filter(category) {\n    currentFilter = category;\n    render();\n  }\n\n  // Setup event listeners\n  function setupListeners() {\n    // Filter buttons\n    const filterBtns = document.querySelectorAll('[data-filter]');\n    filterBtns.forEach(btn => {\n      btn.addEventListener('click', (e) => {\n        const category = e.target.dataset.filter;\n        filterBtns.forEach(b => b.classList.remove('active'));\n        e.target.classList.add('active');\n        filter(category);\n      });\n    });\n\n    // Refresh button (optional)\n    const refreshBtn = document.getElementById('refresh-berita');\n    if (refreshBtn) {\n      refreshBtn.addEventListener('click', async () => {\n        await loadBerita();\n        showToast('Berita diperbarui', 'success');\n      });\n    }\n  }\n\n  return {\n    init,\n    loadBerita,\n    getBerita,\n    filter\n  };\n})();\n\n// Initialize when DOM ready\nif (document.readyState === 'loading') {\n  document.addEventListener('DOMContentLoaded', () => {\n    BeritaModule.init();\n  });\n} else {\n  BeritaModule.init();\n}\n```\n\n### HTML Usage\n\nDi file `berita.html`:\n\n```html\n<!-- Supabase Library -->\n<script ========================================== */\n\nconst DiskusiModule = (() => {\n  let diskusiData = [];\n\n  // Initialize\n  async function init() {\n    try {\n      console.log('🔄 Initializing Diskusi module...');\n      \n      if (!window.supabaseClient) {\n        throw new Error('Supabase client not initialized');\n      }\n\n      // Load existing diskusi\n      await loadDiskusi();\n      \n      // Setup form\n      setupForm();\n      \n      console.log('✅ Diskusi module initialized');\n    } catch (error) {\n      console.error('❌ Error initializing Diskusi:', error);\n      showToast('Error loading diskusi', 'error');\n    }\n  }\n\n  // Load diskusi dari Supabase\n  async function loadDiskusi() {\n    try {\n      console.log('📥 Loading diskusi...');\n      \n      const { data, error } = await window.supabaseClient\n        .from('diskusi')\n        .select('*')\n        .order('date', { ascending: false });\n\n      if (error) throw error;\n\n      diskusiData = data || [];\n      console.log(`✅ Loaded ${diskusiData.length} diskusi topics`);\n      \n      render();\n      return diskusiData;\n    } catch (error) {\n      console.error('❌ Error loading diskusi:', error);\n      return [];\n    }\n  }\n\n  // Render diskusi ke halaman\n  function render() {\n    try {\n      const container = document.getElementById('diskusi-container');\n      if (!container) return;\n\n      if (diskusiData.length === 0) {\n        container.innerHTML = '<p class=\"text-center text-muted\">Belum ada diskusi. Mulai yang pertama!</p>';\n        return;\n      }\n\n      container.innerHTML = diskusiData\n        .map(item => `\n          <div class=\"diskusi-card\">\n            <h3>${item.title}</h3>\n            <p class=\"author\">Oleh: ${item.author || 'Anonim'}</p>\n            <p class=\"content\">${item.content}</p>\n            <p class=\"meta\">\n              <span class=\"date\">${new Date(item.date).toLocaleDateString('id-ID')}</span>\n              <span class=\"comments\">${item.comments || 0} komentar</span>\n            </p>\n          </div>\n        `)\n        .join('');\n    } catch (error) {\n      console.error('❌ Error rendering diskusi:', error);\n    }\n  }\n\n  // Submit diskusi baru\n  async function submitDiskusi(formData) {\n    try {\n      console.log('📤 Submitting diskusi...');\n      \n      // Validate\n      if (!formData.author || !formData.title || !formData.content) {\n        throw new Error('Semua field harus diisi');\n      }\n\n      // Insert ke Supabase\n      const { data, error } = await window.supabaseClient\n        .from('diskusi')\n        .insert([{\n          author: formData.author,\n          title: formData.title,\n          content: formData.content,\n          date: new Date().toISOString().split('T')[0],\n          comments: 0\n        }])\n        .select();\n\n      if (error) throw error;\n\n      console.log('✅ Diskusi submitted:', data);\n      showToast('Diskusi berhasil dibuat!', 'success');\n      \n      // Reload dan render\n      await loadDiskusi();\n      \n      // Clear form\n      const form = document.getElementById('diskusi-form');\n      if (form) form.reset();\n      \n      return data[0];\n    } catch (error) {\n      console.error('❌ Error submitting diskusi:', error);\n      showToast('Gagal membuat diskusi: ' + error.message, 'error');\n      throw error;\n    }\n  }\n\n  // Setup form listener\n  function setupForm() {\n    const form = document.getElementById('diskusi-form');\n    if (!form) return;\n\n    form.addEventListener('submit', async (e) => {\n      e.preventDefault();\n      \n      const formData = {\n        author: form.querySelector('input[name=\"author\"]').value,\n        title: form.querySelector('input[name=\"title\"]').value,\n        content: form.querySelector('textarea[name=\"content\"]').value\n      };\n\n      await submitDiskusi(formData);\n    });\n  }\n\n  return {\n    init,\n    loadDiskusi,\n    submitDiskusi\n  };\n})();\n\n// Initialize when DOM ready\nif (document.readyState === 'loading') {\n  document.addEventListener('DOMContentLoaded', () => {\n    DiskusiModule.init();\n  });\n} else {\n  DiskusiModule.init();\n}\n```\n\n### HTML Usage\n\nDi file
